1//! Building a site from its pages.
2//!
3//! [`Site::load`] reads every page's front matter, and the site's
4//! [`Theme`] builds it with [`Theme::build`]:
5//!
6//! 1. Pages with template tags are rendered by [`Theme::template`] first,
7//!    after the others, so they can list them with their word counts.
8//! 2. Every page's Markdown is rendered to HTML by [`Markdown`].
9//! 3. The page is wrapped in its layout, by [`Theme::layout`], and then in
10//!    the document around it, by [`Theme::document`].
11//! 4. The Markdown, wrapped in its layout but not the document, is
12//!    published next to the HTML, like `/posts/foo.md`.
13//!
14//! Everything else in the crate that's part of the site, like images and
15//! CSS, is copied over as it is.
16

31/// A site and its pages. `E` is the site's own front matter, see
32/// [`FrontMatter`].
33pub struct Site<E = IgnoredAny> {
34    pub config: &'static Config,
35    pub options: Options,
36    /// When the build started, which dates undated pages.
37    pub time: DateTime<Local>,
38    /// Every page, those in collections sorted by date, then by path.
39    pub pages: Vec<Page<E>>,
40}
41
42/// How to build a site.
43#[derive(Debug, Clone)]
44pub struct Options {
45    /// The top of the site's crate, with the files copied into the site.
46    pub source: PathBuf,
47    /// Where the site is written, cleared first.
48    pub destination: PathBuf,
49    /// What every path on the site starts with, for previews under a
50    /// path, like `/pr-preview/pr-83`.
51    pub baseurl: String,
52    /// Build pages marked `published: false` too.
53    pub unpublished: bool,
54}
55
56/// What a site makes of its pages: its templates and layouts, and any
57/// files of its own. A site implements it on a type of its own holding its
58/// [`Site`], and anything else its templates need, and builds with
59/// [`Theme::build`].
60pub trait Theme: Sized {
61    /// The site's own front matter fields, see [`FrontMatter`].
62    type Extra: DeserializeOwned + Default;
63
64    fn site(&self) -> &Site<Self::Extra>;
65
66    fn site_mut(&mut self) -> &mut Site<Self::Extra>;
67
68    /// The page's Markdown with its template tags rendered, for a page
69    /// with them, front matter and all. The build script makes an askama
70    /// template for each such page, so this is usually its generated
71    /// `render`.
72    fn template(&self, page: &Page<Self::Extra>) -> Option<Result<String, String>>;
73
74    /// `content`, the page's HTML or its Markdown to publish, wrapped in
75    /// its layout, like a title and date above it. Unchanged by default.
76    fn layout(&self, page: &Page<Self::Extra>, content: &str) -> Result<String, String> {
77        let _ = page;
78        Ok(content.to_owned())
79    }
80
81    /// The page's HTML, already in its layout, as a whole document, with
82    /// the site's header and footer. Unchanged by default.
83    fn document(&self, page: &Page<Self::Extra>, content: &str) -> Result<String, String> {
84        let _ = page;
85        Ok(content.to_owned())
86    }
87
88    /// The front matter the page's Markdown is published with.
89    fn front_matter(&self, page: &Page<Self::Extra>) -> Yaml {
90        let title = page.title();
91        let date = page.date.map(|d| d.format("%Y-%m-%d").to_string());
92        Yaml::new()
93            .string("title", (!title.is_empty()).then_some(title.as_str()))
94            .string("date", date.as_deref())
95            .string("description", page.front.description.as_deref())
96            .flag("draft", page.front.draft)
97    }
98
99    /// More files to write, from the top of the site, like a feed or an
100    /// index of the pages.
101    fn files(&self) -> Result<Vec<(String, Vec<u8>)>, String> {
102        Ok(Vec::new())
103    }
104
105    /// Renders every page and writes the site.
106    fn build(&mut self) -> Result<(), String> {
107        self.render()?;
108        self.write()
109    }
110
111    /// Renders the templates, then the Markdown, of every page.
112    #[doc(hidden)]
113    fn render(&mut self) -> Result<(), String> {
114        let site = self.site();
115        let highlighter = Highlighter::default();
116        let tikz = Tikz::new(
117            site.options.source.join(".site-cache/tikz"),
118            site.config.tikz_preamble,
119        );
120        let markdown = Markdown {
121            highlighter: &highlighter,
122            tikz: &tikz,
123        };
124        let render = |page: &mut Page<Self::Extra>| -> Result<(), String> {
125            if !page.is_markdown() {
126                return Ok(());
127            }
128            page.content = markdown
129                .render(&page.markdown)
130                .map_err(|e| format!("{}: {e}", page.source.path))?;
131            if let Some(excerpt) = page.excerpt_markdown() {
132                page.excerpt = markdown.render(&excerpt)?;
133            }
134            Ok(())
135        };
136        for page in self.site_mut().pages.iter_mut() {
137            if !page.is_template() {
138                render(page)?;
139            }
140        }
141        let mut rendered = Vec::new();
142        for (i, page) in self.site().pages.iter().enumerate() {
143            if let Some(text) = self.template(page) {
144                let text = text.map_err(|e| format!("{}: {e}", page.source.path))?;
145                rendered.push((i, FrontMatter::split(&text).1.to_owned()));
146            }
147        }
148        for (i, text) in rendered {
149            let page = &mut self.site_mut().pages[i];
150            page.markdown = text;
151            render(page)?;
152        }
153        Ok(())
154    }
155
156    /// Writes the site into the destination, which is cleared first, all
157    /// but what the config says to keep.
158    #[doc(hidden)]
159    fn write(&self) -> Result<(), String> {
160        let site = self.site();
161        site.clear()?;
162        site.copy_files(&site.options.source)?;
163        for page in &site.pages {
164            let out = if page.is_markdown() {
165                let content = self.layout(page, &page.content)?;
166                match page.front.layout.as_deref() {
167                    None | Some("null") => content,
168                    Some(_) => self.document(page, &content)?,
169                }
170            } else {
171                page.markdown.clone()
172            };
173            site.write_file(&page.output_path(), site.with_baseurl(&out))?;
174            if let Some(url) = page.markdown_url() {
175                let body = self.layout(page, &page.markdown)?;
176                let published = self.front_matter(page).with_body(&body);
177                site.write_file(&url, site.with_baseurl(&published))?;
178            }
179        }
180        for (path, contents) in self.files()? {
181            site.write_file(&path, contents)?;
182        }
183        // The site is already built, so stop GitHub Pages from running
184        // Jekyll on it.
185        site.write_file(".nojekyll", "")
186    }
187}

189impl<E: DeserializeOwned + Default> Site<E> {
190    /// Reads every page in `sources`, the build script's `SOURCES`, ready
191    /// to build.
192    pub fn load(
193        config: &'static Config,
194        sources: &[&'static Source],
195        options: Options,
196    ) -> Result<Self, String> {
197        let time = Local::now();
198        let mut pages = Vec::new();
199        for source in sources {
200            let page = Page::read(config, source)?;
201            if page.front.published || options.unpublished {
202                pages.push(page);
203            }
204        }
205        // Undated pages sort as if written now.
206        let today = time.date_naive();
207        pages.sort_by(|a, b| {
208            let key = |p: &Page<E>| (p.date.unwrap_or(today), p.source.path);
209            key(a).cmp(&key(b))
210        });
211        Ok(Self {
212            config,
213            options,
214            time,
215            pages,
216        })
217    }
218}

220impl<E> Site<E> {
221    /// The pages in the collection `label`, by date.
222    pub fn collection(&self, label: &str) -> Vec<&Page<E>> {
223        self.pages
224            .iter()
225            .filter(|p| p.collection.is_some_and(|c| c.label == label))
226            .collect()
227    }
228
229    /// The pages in the collection `label`, newest first.
230    pub fn newest_first(&self, label: &str) -> Vec<&Page<E>> {
231        let mut pages = self.collection(label);
232        pages.reverse();
233        pages
234    }
235
236    /// `path`, from the site's root, with the base URL in front.
237    pub fn relative_url(&self, path: &str) -> String {
238        format!("{}{path}", self.options.baseurl)
239    }
240
241    /// `path`, from the site's root, as a full URL.
242    pub fn absolute_url(&self, path: &str) -> String {
243        format!("{}{}{path}", self.config.url, self.options.baseurl)
244    }
245
246    /// The build time, like `2026-10-30T17:03:50+00:00`.
247    pub fn xml_time(&self) -> String {
248        self.time
249            .to_rfc3339_opts(chrono::SecondsFormat::Secs, false)
250    }
251
252    /// The build time, for adding to URLs so browsers fetch new copies.
253    pub fn build_id(&self) -> i64 {
254        self.time.timestamp()
255    }
256
257    /// The page's date, at midnight where the site's built, for feeds.
258    pub fn xml_date(&self, page: &Page<E>) -> String {
259        let date = page.date.unwrap_or(self.time.date_naive());
260        match date
261            .and_hms_opt(0, 0, 0)
262            .and_then(|d| d.and_local_timezone(Local).single())
263        {
264            Some(time) => time.to_rfc3339_opts(chrono::SecondsFormat::Secs, false),
265            None => date.to_string(),
266        }
267    }
268
269    /// `text` escaped for HTML or XML.
270    pub fn escape(&self, text: &str) -> String {
271        Text::escape(text)
272    }
273
274    /// Writes `contents` to `path`, from the top of the destination.
275    pub fn write_file(&self, path: &str, contents: impl AsRef<[u8]>) -> Result<(), String> {
276        let file = self.options.destination.join(path.trim_start_matches('/'));
277        fs::create_dir_all(file.parent().unwrap()).map_err(|e| e.to_string())?;
278        fs::write(&file, contents).map_err(|e| format!("{}: {e}", file.display()))
279    }
280
281    /// Empties the destination, all but what the config says to keep.
282    fn clear(&self) -> Result<(), String> {
283        let Ok(entries) = fs::read_dir(&self.options.destination) else {
284            return Ok(());
285        };
286        for entry in entries.flatten() {
287            let path = entry.path();
288            if self
289                .config
290                .keep
291                .iter()
292                .any(|keep| entry.file_name() == *keep)
293            {
294                continue;
295            }
296            let removed = if path.is_dir() {
297                fs::remove_dir_all(&path)
298            } else {
299                fs::remove_file(&path)
300            };
301            removed.map_err(|e| format!("{}: {e}", path.display()))?;
302        }
303        Ok(())
304    }
305
306    /// Copies every file in `dir` that's part of the site, but isn't a
307    /// page, into the same place in the destination.
308    fn copy_files(&self, dir: &Path) -> Result<(), String> {
309        let entries = fs::read_dir(dir).map_err(|e| format!("{}: {e}", dir.display()))?;
310        for entry in entries.flatten() {
311            let path = entry.path();
312            let relative = path
313                .strip_prefix(&self.options.source)
314                .unwrap()
315                .to_string_lossy()
316                .replace('\\', "/");
317            if self.config.ignored(&relative)
318                || self.pages.iter().any(|p| p.source.path == relative)
319            {
320                continue;
321            }
322            if path.is_dir() {
323                self.copy_files(&path)?;
324            } else {
325                let to = self.options.destination.join(&relative);
326                fs::create_dir_all(to.parent().unwrap()).map_err(|e| e.to_string())?;
327                fs::copy(&path, &to).map_err(|e| format!("{relative}: {e}"))?;
328            }
329        }
330        Ok(())
331    }
332
333    /// The text with the base URL put in front of every link and source
334    /// from the site's root, like `href="/about"`, Markdown links like
335    /// `](/about)` included. Code blocks are left alone.
336    fn with_baseurl(&self, text: &str) -> String {
337        let base = &self.options.baseurl;
338        if base.is_empty() {
339            return text.to_owned();
340        }
341        let mut out = String::with_capacity(text.len());
342        let mut fence: Option<&str> = None;
343        for line in text.split_inclusive('\n') {
344            let trimmed = line.trim_start();
345            if let Some(marker) = fence {
346                if trimmed.starts_with(marker) {
347                    fence = None;
348                }
349                out.push_str(line);
350                continue;
351            }
352            if let Some(marker) = ["```", "~~~"].into_iter().find(|m| trimmed.starts_with(m)) {
353                fence = Some(marker);
354                out.push_str(line);
355                continue;
356            }
357            let mut line = line.to_owned();
358            for prefix in ["href=\"", "src=\"", "data-music-xml=\"", "](", "]: "] {
359                line = Self::prefix_paths(&line, prefix, base);
360            }
361            out.push_str(&line);
362        }
363        out
364    }
365
366    /// Puts `base` in front of every path from the root after `prefix`.
367    fn prefix_paths(text: &str, prefix: &str, base: &str) -> String {
368        let mut out = String::with_capacity(text.len());
369        let mut rest = text;
370        while let Some(i) = rest.find(prefix) {
371            let after = i + prefix.len();
372            out.push_str(&rest[..after]);
373            rest = &rest[after..];
374            if rest.starts_with('/') && !rest.starts_with("//") && !rest.starts_with(base) {
375                out.push_str(base);
376            }
377        }
378        out.push_str(rest);
379        out
380    }
