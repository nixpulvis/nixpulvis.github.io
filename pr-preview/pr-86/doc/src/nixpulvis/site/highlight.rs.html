1//! Code blocks, highlighted with syntect's classes, which `css/syntax.css`
2//! colors.
3
4use syntect::html::{ClassStyle, ClassedHTMLGenerator};
5use syntect::parsing::{SyntaxReference, SyntaxSet};
6use syntect::util::LinesWithEndings;
7
8use super::text::Text;
9
10/// Highlights code blocks.
11pub struct Highlighter {
12    syntaxes: SyntaxSet,
13}
14
15impl Default for Highlighter {
16    fn default() -> Self {
17        Self {
18            syntaxes: SyntaxSet::load_defaults_newlines(),
19        }
20    }
21}
22
23impl Highlighter {
24    /// Languages left uncolored, but shown like highlighted code.
25    const PLAIN: &[&str] = &["plaintext", "text", "conf", "elixir"];
26
27    /// The HTML for a code block in `lang`, which may be empty.
28    ///
29    /// Languages syntect knows, and plain text, are highlighted inside
30    /// `<div class="language-rust"><div class="highlight">`, like Jekyll's
31    /// rouge did. Code in any other language is left in a plain `<pre>`.
32    pub fn block(&self, lang: &str, code: &str) -> String {
33        let lang = if lang.is_empty() { "plaintext" } else { lang };
34        let syntax = match self.syntax(lang) {
35            Some(syntax) => syntax,
36            None if Self::PLAIN.contains(&lang) => self.syntaxes.find_syntax_plain_text(),
37            None => {
38                return format!(
39                    "<pre><code class=\"language-{lang}\">{}</code></pre>\n",
40                    Text::escape(code)
41                );
42            }
43        };
44        let mut html =
45            ClassedHTMLGenerator::new_with_class_style(syntax, &self.syntaxes, ClassStyle::Spaced);
46        for line in LinesWithEndings::from(code) {
47            // Only fails on a broken syntax definition.
48            let _ = html.parse_html_for_line_which_includes_newline(line);
49        }
50        format!(
51            "<div class=\"language-{lang}\"><div class=\"highlight\"><pre class=\"highlight\"><code>{}</code></pre></div></div>\n",
52            html.finalize()
53        )
54    }
55
56    fn syntax(&self, lang: &str) -> Option<&SyntaxReference> {
57        let token = match lang {
58            "plaintext" | "text" => return None,
59            // Close enough, and better than nothing.
60            "racket" | "scheme" => "lisp",
61            "rb" => "ruby",
62            lang => lang,
63        };
64        self.syntaxes.find_syntax_by_token(token)
65    }
66}
