1//! The layouts pages are wrapped in, from the askama templates in
2//! `templates/`.
3//!
4//! A page names its layout in its front matter. Every layout but
5//! [`DefaultLayout`] wraps the page's content in a title and the like, and is
6//! written in Markdown, so the published Markdown gets it too. Then
7//! [`DefaultLayout`] wraps that in the site's header, nav and footer.
8
9use askama::Template;
10
11use super::text::Text;
12use super::{Page, Site};
13
14/// A page's layout.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub enum Layout {
17    /// Only the site's header and footer.
18    Default,
19    /// The title and word count, for mathematics.
20    Article,
21    /// The title and date, for notes.
22    Note,
23    /// The title as the page's heading, for index pages like Ramblings.
24    Page,
25    /// The title, status and links, for projects.
26    Project,
27    /// The title, word count and date, for ramblings.
28    Rambling,
29    /// Nothing at all, for pages like `feed.xml`.
30    None,
31}
32
33impl Layout {
34    /// The layout called `name` in a page's front matter.
35    pub fn named(name: Option<&str>) -> Self {
36        match name {
37            Some("default") => Self::Default,
38            Some("article") => Self::Article,
39            Some("note") => Self::Note,
40            Some("page") => Self::Page,
41            Some("project") => Self::Project,
42            Some("rambling") => Self::Rambling,
43            _ => Self::None,
44        }
45    }
46
47    /// `content`, the page's HTML or Markdown, wrapped in the layout, but
48    /// not in [`DefaultLayout`].
49    pub fn wrap(self, site: &Site, page: &Page, content: &str) -> Result<String, String> {
50        let wrapped = match self {
51            Self::Default | Self::None => return Ok(content.to_owned()),
52            Self::Article => Article {
53                site,
54                page,
55                content,
56            }
57            .render(),
58            Self::Note => Note {
59                site,
60                page,
61                content,
62            }
63            .render(),
64            Self::Page => PageLayout {
65                site,
66                page,
67                content,
68            }
69            .render(),
70            Self::Project => Project {
71                site,
72                page,
73                content,
74            }
75            .render(),
76            Self::Rambling => Rambling {
77                site,
78                page,
79                content,
80            }
81            .render(),
82        };
83        wrapped.map_err(|e| format!("{}: {e}", page.source.path))
84    }
85}

87/// The site's header, nav and footer around every page.
88#[derive(Template)]
89#[template(path = "default.html", escape = "none")]
90pub struct DefaultLayout<'a> {
91    pub site: &'a Site,
92    pub page: &'a Page,
93    /// The page's HTML, in its own layout.
94    pub content: &'a str,
95}
96
97impl DefaultLayout<'_> {
98    /// The page's title, or the name of its file.
99    fn title(&self) -> String {
100        let title = self.page.title();
101        if title.is_empty() {
102            self.page
103                .source
104                .path
105                .rsplit('/')
106                .next()
107                .unwrap_or("")
108                .to_owned()
109        } else {
110            title
111        }
112    }
113
114    /// A summary for search engines and link previews: the page's
115    /// description, or its first paragraph, or the site's description.
116    fn description(&self) -> String {
117        let excerpt = Text::normalize_whitespace(&Text::strip_html(&self.page.excerpt));
118        let excerpt = excerpt.trim();
119        let description = match &self.page.front.description {
120            Some(description) => description.as_str(),
121            None if !excerpt.is_empty() => excerpt,
122            None => Site::DESCRIPTION,
123        };
124        let description = Text::normalize_whitespace(&Text::strip_html(description));
125        Text::escape_once(&Text::truncate(&description, 160))
126    }
127
128    /// Whether MathJax is needed, for `\(` or `\[` math.
129    fn math(&self) -> bool {
130        self.content.contains("\\(") || self.content.contains("\\[")
131    }
132
133    /// A link to open an issue about this page.
134    fn issue_url(&self) -> String {
135        format!(
136            "{}/issues/new?body={}",
137            Site::REPOSITORY,
138            Text::url_encode(&self.site.absolute_url(&self.page.url))
139        )
140    }
141}

143macro_rules! layout {
144    ($name:ident, $path:literal) => {
145        #[derive(Template)]
146        #[template(path = $path, escape = "none")]
147        struct $name<'a> {
148            #[allow(dead_code)]
149            site: &'a Site,
150            page: &'a Page,
151            content: &'a str,
152        }
153    };
154}
