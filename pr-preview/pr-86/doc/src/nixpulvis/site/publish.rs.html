1//! Each page's Markdown, published next to its HTML for the terminal view
2//! and anyone else who'd rather read it, like `/ramblings/foo.md`.
3//!
4//! It's the Markdown after its template tags are rendered, wrapped in the
5//! page's layout, but not the site's default one with the header and nav.
6//! Its front matter keeps what a reader might want, and adds the word
7//! count and collection.
8
9use std::fmt;
10
11use super::layout::Layout;
12use super::{Page, Site};
13
14/// A page's published Markdown.
15pub struct Published {
16    front: PublishedFrontMatter,
17    body: String,
18}
19
20/// The front matter a page is published with.
21#[derive(Debug, Default)]
22pub struct PublishedFrontMatter {
23    pub title: Option<String>,
24    /// Like `2026-10-19`, only for pages with a date of their own.
25    pub date: Option<String>,
26    pub description: Option<String>,
27    pub draft: bool,
28    pub status: Option<String>,
29    pub references: Vec<String>,
30    /// For layouts that show a word count.
31    pub words: Option<usize>,
32    pub collection: Option<String>,
33}
34
35impl Published {
36    pub fn new(site: &Site, page: &Page) -> Result<Self, String> {
37        let layout = Layout::named(page.front.layout.as_deref());
38        Ok(Self {
39            front: Self::front_matter(site, page),
40            body: layout.wrap(site, page, &page.markdown)?,
41        })
42    }
43
44    pub fn front_matter(_site: &Site, page: &Page) -> PublishedFrontMatter {
45        let title = page.title();
46        PublishedFrontMatter {
47            title: (!title.is_empty()).then_some(title),
48            date: page.date.map(|d| d.format("%Y-%m-%d").to_string()),
49            description: page.front.description.clone(),
50            draft: page.front.draft,
51            status: page.front.status.clone(),
52            references: page.front.references.clone(),
53            words: matches!(page.front.layout.as_deref(), Some("rambling" | "article"))
54                .then(|| page.words()),
55            collection: page.collection.map(|c| c.label.to_owned()),
56        }
57    }
58}
59
60impl fmt::Display for Published {
61    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
62        let front = &self.front;
63        let mut yaml = String::new();
64        let mut field = |key: &str, value: Option<String>| {
65            if let Some(value) = value {
66                yaml.push_str(&format!("{key}: {value}\n"));
67            }
68        };
69        field("title", front.title.as_deref().map(Yaml::string));
70        field("date", front.date.as_deref().map(Yaml::string));
71        field(
72            "description",
73            front.description.as_deref().map(Yaml::string),
74        );
75        field("draft", front.draft.then(|| "true".to_owned()));
76        field("status", front.status.as_deref().map(Yaml::string));
77        if !front.references.is_empty() {
78            yaml.push_str("references:\n");
79            for reference in &front.references {
80                yaml.push_str(&format!("- {}\n", Yaml::string(reference)));
81            }
82        }
83        let mut field = |key: &str, value: Option<String>| {
84            if let Some(value) = value {
85                yaml.push_str(&format!("{key}: {value}\n"));
86            }
87        };
88        field("words", front.words.map(|w| w.to_string()));
89        field("collection", front.collection.as_deref().map(Yaml::string));
90        if yaml.is_empty() {
91            f.write_str(&self.body)
92        } else {
93            write!(f, "---\n{yaml}---\n\n{}", self.body)
94        }
95    }
96}

98/// Writing YAML the way Ruby's Psych does, which published it before.
99struct Yaml;
100
101impl Yaml {
102    /// The string plain, or in single quotes where plain would read as
103    /// something else.
104    fn string(s: &str) -> String {
105        let looks_like_other = s.is_empty()
106            || s.parse::<f64>().is_ok()
107            || ["true", "false", "yes", "no", "on", "off", "null", "~"]
108                .contains(&s.to_ascii_lowercase().as_str())
109            || (s.len() >= 10
110                && s.as_bytes()[4] == b'-'
111                && s[..4].bytes().all(|b| b.is_ascii_digit()));
112        let special = s.contains(": ")
113            || s.contains(" #")
114            || s.ends_with(':')
115            || s.starts_with(char::is_whitespace)
116            || s.ends_with(char::is_whitespace)
117            || s.starts_with([
118                '-', '?', ':', ',', '[', ']', '{', '}', '#', '&', '*', '!', '|', '>', '\'', '"',
119                '%', '@', '`',
120            ]);
121        if looks_like_other || special {
122            format!("'{}'", s.replace('\'', "''"))
123        } else {
124            s.to_owned()
125        }
126    }
127}
