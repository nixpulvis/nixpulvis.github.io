1//! Browse nixpulvis.com as a terminal app.
2//!
3//! The app reads the site's published Markdown and `site.json`, renders
4//! pages with ratmark, and draws them with Ratatui. It never does I/O itself.
5//! Methods return [`Effect`]s for the host to carry out, which keeps the
6//! same code working in a real terminal and in the browser.
7

26/// How long a notice shows before the key hints come back.
27const NOTICE_TIME: Duration = Duration::from_secs(3);

29/// Widest the page's text gets, for comfortable reading.
30const MAX_WIDTH: u16 = 84;

32/// How many columns `<` and `>` shift wide code by.
33const SHIFT: isize = 8;

35/// Something for the host to do.
36#[derive(Debug, Clone, PartialEq, Eq)]
37pub enum Effect {
38    /// Fetch `site.json` from this path and pass it to [`App::site_loaded`].
39    FetchSite(String),
40    /// Fetch a page's Markdown from `source` and pass it to
41    /// [`App::page_loaded`] along with `path`.
42    FetchPage {
43        path: String,
44        source: String,
45    },
46    /// Open a link that isn't part of the site.
47    Open(String),
48    /// Leave for another page of the site, at this path, which the host
49    /// loads and starts the app on again. Only sent after
50    /// [`App::with_host_history`].
51    Navigate(String),
52    /// Go back to the page before this one, which the host remembers. Only
53    /// sent after [`App::with_host_history`].
54    Back,
55    Quit,
56}
57
58/// Input, already translated from whatever the host's events look like.
59#[derive(Debug, Clone, Copy, PartialEq, Eq)]
60pub enum Key {
61    Up,
62    Down,
63    PageUp,
64    PageDown,
65    Home,
66    End,
67    Tab,
68    BackTab,
69    Enter,
70    Back,
71    Esc,
72    Char(char),
73}

75struct Page {
76    path: String,
77    /// The whole page, before any tag filter.
78    markdown: String,
79    title: String,
80    /// When it was published or drafted, like `October 19, 2026`.
81    date: Option<String>,
82    draft: bool,
83    /// The word count, on pages whose layout shows one.
84    words: Option<String>,
85    /// A project's status, like `complete`.
86    status: Option<String>,
87    view: MarkdownState,
88}

90impl Page {
91    /// The line under a post, worded like the site's, as in `7121 words
92    /// published on October 19, 2026`, or a shorter wording when that's
93    /// wider than `width`, dropping the word count first.
94    fn meta(&self, width: usize) -> Option<String> {
95        let verb = if self.draft { "drafted" } else { "published" };
96        let mut wordings = Vec::new();
97        if let Some(words) = &self.words {
98            wordings.push(match &self.date {
99                Some(date) => format!("{words} words {verb} on {date}"),
100                None => format!("{words} words"),
101            });
102        }
103        if let Some(date) = &self.date {
104            wordings.push(format!(
105                "{}{} on {date}",
106                verb[..1].to_uppercase(),
107                &verb[1..]
108            ));
109        }
110        if self.draft {
111            wordings.push("Draft".to_owned());
112        }
113        wordings
114            .into_iter()
115            .find(|w| Span::raw(w.as_str()).width() <= width)
116    }
117}

188    /// Starts on the page at `path`. `base` is the path the site lives
189    /// under, `/` except in previews.
190    pub fn start(&mut self, base: &str, path: &str) -> Vec<Effect> {
191        let base = if base.ends_with('/') {
192            base.to_owned()
193        } else {
194            format!("{base}/")
195        };
196        self.base = base.trim_end_matches('/').to_owned();
197        self.not_found = format!("{base}404.md");
198        let mut effects = vec![Effect::FetchSite(format!("{base}site.json"))];
199        effects.extend(self.go(page_target(path), false));
200        effects
201    }

203    pub fn site_loaded(&mut self, json: Result<String, String>) {
204        match json.and_then(|j| serde_json::from_str::<Index>(&j).map_err(|e| e.to_string())) {
205            Ok(mut site) => {
206                site.nav.retain(|n| n.tui);
207                self.site = Some(site);
208            }
209            Err(e) => self.status = Some(format!("Couldn't load the site index: {e}")),
210        }
211    }

213    pub fn page_loaded(&mut self, path: &str, markdown: Result<String, String>) -> Vec<Effect> {
214        if self.loading.as_deref() != Some(path) {
215            return Vec::new();
216        }
217        self.loading = None;
218        let not_found = std::mem::take(&mut self.showing_not_found);
219        match markdown {
220            Ok(markdown) => {
221                // The site's 404 names the missing page with a script.
222                let markdown = if not_found {
223                    markdown.replace(">that-page<", &format!(">{path}<"))
224                } else {
225                    markdown
226                };
227                let front = ratmark::front_matter::FrontMatter::split(&markdown).0;
228                let get = |key| front.as_ref().and_then(|f| f.get(key));
229                let title = get("title").unwrap_or_else(|| path.to_owned());
230                let date = get("date").map(|d| long_date(&d).unwrap_or(d));
231                let draft = get("draft").is_some_and(|d| d == "true");
232                let words = get("words");
233                let status = get("status");
234                let markdown = draft_prefixes(&status_after_title(&without_title(&markdown)));
235                let mut view = self.view(path, markdown.clone());
236                view.scroll_to(self.restore_scroll.take().unwrap_or(0));
237                if let Some(anchor) = self.pending_anchor.take() {
238                    view.scroll_to_anchor(&anchor);
239                }
240                self.page = Some(Page {
241                    path: path.to_owned(),
242                    markdown,
243                    title,
244                    date,
245                    draft,
246                    words,
247                    status,
248                    view,
249                });
250                self.status = None;
251                Vec::new()
252            }
253            Err(_) if !not_found => {
254                self.loading = Some(path.to_owned());
255                self.showing_not_found = true;
256                vec![Effect::FetchPage {
257                    path: path.to_owned(),
258                    source: self.not_found.clone(),
259                }]
260            }
261            Err(e) => {
262                self.status = Some(format!("Couldn't load {path}: {e}"));
263                Vec::new()
264            }
265        }
266    }

268    /// The view of `markdown`, the page at `path`.
269    // Without native links, the page's address goes unused.

278    fn go(&mut self, target: Target, remember: bool) -> Vec<Effect> {
279        match target {
280            Target::Page { path, anchor } => {
281                if remember {
282                    self.restore_scroll = None;
283                    match (&mut self.history, &self.page) {
284                        (Some(history), Some(page)) => {
285                            history.push((page.path.clone(), page.view.scroll()));
286                        }
287                        (Some(_), None) => {}
288                        // The host loads the page, as it would any link.
289                        (None, _) => {
290                            self.status = Some(format!("Loading {path}…"));
291                            return vec![Effect::Navigate(match anchor {
292                                Some(anchor) => format!("{path}#{anchor}"),
293                                None => path,
294                            })];
295                        }
296                    }
297                }
298                self.loading = Some(path.clone());
299                self.showing_not_found = false;
300                self.pending_anchor = anchor;
301                self.status = Some(format!("Loading {path}…"));
302                vec![Effect::FetchPage {
303                    source: links::source(&path),
304                    path,
305                }]
306            }
307            Target::Anchor(anchor) => {
308                let Some(page) = &self.page else {
309                    return Vec::new();
310                };
311                // On a page of tagged items, a tag's link picks the items
312                // with that tag, which the site does with CSS.
313                if let Some(markdown) = filter_tagged(&page.markdown, &anchor) {
314                    let (path, scroll) = (page.path.clone(), page.view.scroll());
315                    let mut view = self.view(&path, markdown);
316                    view.scroll_to(scroll);
317                    if let Some(page) = &mut self.page {
318                        page.view = view;
319                    }
320                } else if let Some(page) = &mut self.page {
321                    page.view.scroll_to_anchor(&anchor);
322                }
323                Vec::new()
324            }
325            Target::External(url) => {
326                self.notice = Some((format!("Opened {url}"), Instant::now()));
327                vec![Effect::Open(url)]
328            }
329        }
330    }

333    pub fn follow(&mut self, href: &str) -> Vec<Effect> {
334        let current = self.page.as_ref().map_or("/", |p| p.path.as_str());

339    fn back(&mut self) -> Vec<Effect> {
340        let Some(history) = &mut self.history else {
341            return vec![Effect::Back];
342        };
343        let Some((path, scroll)) = history.pop() else {
344            return Vec::new();
345        };
346        self.restore_scroll = Some(scroll);
347        self.go(page_target(&path), false)
348    }

350    pub fn key(&mut self, key: Key) -> Vec<Effect> {
351        if let Key::Char('q') | Key::Esc = key {
352            return vec![Effect::Quit];
353        }
354        if let Key::Back | Key::Char('h') = key {
355            return self.back();
356        }
357        if let Key::Char(c @ '1'..='9') = key {
358            let i = usize::from(c as u8 - b'1');
359            return match self.nav_paths().get(i).cloned() {
360                Some(path) => self.go(Target::Page { path, anchor: None }, true),
361                None => Vec::new(),
362            };
363        }
364        let Some(page) = &mut self.page else {
365            return Vec::new();
366        };
367        let view = &mut page.view;
368        match key {
369            Key::Down | Key::Char('j') => _ = view.scroll_by(1),
370            Key::Up | Key::Char('k') => _ = view.scroll_by(-1),
371            Key::PageDown | Key::Char(' ') | Key::Char('f') => view.page(true),
372            Key::PageUp | Key::Char('b') => view.page(false),
373            Key::Home | Key::Char('g') => view.scroll_to(0),
374            Key::End | Key::Char('G') => view.scroll_to(usize::MAX),
375            Key::Char('>') => _ = view.shift_by(SHIFT),
376            Key::Char('<') => _ = view.shift_by(-SHIFT),
377            Key::Tab => view.focus_next(),
378            Key::BackTab => view.focus_prev(),
379            Key::Enter => {
380                if let Some(url) = view.focused().map(|l| l.url.clone()) {
381                    return self.follow(&url);
382                }
383            }
384            _ => {}
385        }
386        Vec::new()
387    }

389    /// A click or tap at column `x`, row `y` of the screen.
390    pub fn click(&mut self, x: u16, y: u16) -> Vec<Effect> {
391        if self.close.contains((x, y).into()) {
392            return vec![Effect::Quit];
393        }
394        if let Some((_, path)) = self.nav.iter().find(|(r, _)| r.contains((x, y).into())) {
395            let path = path.clone();
396            return self.go(Target::Page { path, anchor: None }, true);
397        }
398        let link = self.page.as_ref().and_then(|p| p.view.link_at(x, y));
399        match link.map(|l| l.url.clone()) {
400            Some(url) => self.follow(&url),
401            None => Vec::new(),
402        }
403    }

449    pub fn draw(&mut self, frame: &mut Frame) {
450        let area = frame.area();
451        frame.render_widget(
452            Block::new().style(Style::new().bg(BACKGROUND).fg(TEXT)),
453            area,
454        );
455
456        // Errors and notices always show. The key hints only show when they fit whole,
457        // and the page takes their line when they don't.
458        if self
459            .notice
460            .as_ref()
461            .is_some_and(|(_, at)| at.elapsed() > NOTICE_TIME)
462        {
463            self.notice = None;
464        }
465        let hint = self.hint();
466        let footer = match (&self.status, &self.notice) {
467            (Some(status), _) | (None, Some((status, _))) => Some(status.as_str()),
468            (None, None) => {
469                Some(hint.as_str()).filter(|h| Span::raw(*h).width() <= usize::from(area.width))
470            }
471        };
472        let [header, body, bottom] = Layout::vertical([
473            Constraint::Length(1),
474            Constraint::Fill(1),
475            Constraint::Length(u16::from(footer.is_some())),
476        ])
477        .areas(area);
478
479        if let Some(footer) = footer {
480            frame.render_widget(
481                Paragraph::new(footer.to_owned())
482                    .style(Style::new().fg(MUTED))
483                    .centered(),
484                bottom,
485            );
486        }
487        self.draw_header(frame, column(header));
488        self.draw_page(frame, column(body));
489    }

491    fn draw_header(&mut self, frame: &mut Frame, area: Rect) {
492        self.nav.clear();
493        let current = self.page.as_ref().map(|p| p.path.as_str()).unwrap_or("");
494        let mut items = vec![("nixpulvis".to_owned(), format!("{}/", self.base))];
495        if let Some(site) = &self.site {
496            for nav in &site.nav {
497                let label = nav.icon.clone().unwrap_or_else(|| nav.title.clone());
498                items.push((label, format!("{}{}", self.base, nav.url)));
499            }
500        }
501        let mut spans = Vec::new();
502        let mut x = area.x + 1;
503        spans.push(Span::raw(" "));
504        for (i, (label, path)) in items.into_iter().enumerate() {
505            let width = Span::raw(&label).width() as u16;
506            let style = if i == 0 {
507                Style::new().fg(TEXT).bold()
508            } else if same_page(current, &path) {
509                Style::new().fg(YELLOW).add_modifier(Modifier::UNDERLINED)
510            } else {
511                Style::new().fg(MUTED)
512            };
513            spans.push(Span::styled(label, style));
514            spans.push(Span::raw("  "));
515            self.nav.push((Rect::new(x, area.y, width, 1), path));
516            x += width + 2;
517        }
518        frame.render_widget(Line::from(spans), area);
519        #[cfg(feature = "native-links")]
520        for (rect, path) in &self.nav {
521            let url = format!("{}{path}", self.origin.trim_end_matches('/'));
522            let link = ratatui::buffer::Hyperlink::new(url);
523            frame.buffer_mut().set_hyperlink(*rect, Some(&link));
524        }
525
526        // A way out for touch screens, with no keyboard for `q`.
527        self.close = Rect::new(area.right().saturating_sub(3), area.y, 3, 1);
528        frame.render_widget(Line::from(" × ").style(Style::new().fg(MUTED)), self.close);
529        // A link, so it looks clickable, to this page as plain HTML. A plain
530        // click leaves, like `q`.
531        #[cfg(feature = "native-links")]
532        {
533            let url = format!("{}{current}", self.origin.trim_end_matches('/'));
534            let link = ratatui::buffer::Hyperlink::new(url);
535            frame.buffer_mut().set_hyperlink(self.close, Some(&link));
536        }
537    }

539    fn draw_page(&mut self, frame: &mut Frame, column: Rect) {
540        let title = self
541            .page
542            .as_ref()
543            .map_or_else(String::new, |p| p.title.clone());
544        // A page the nav names in words is already underlined there.
545        let named_in_nav = self
546            .page
547            .as_ref()
548            .zip(self.site.as_ref())
549            .is_some_and(|(p, site)| {
550                site.nav.iter().any(|nav| {
551                    nav.icon.is_none() && same_page(&p.path, &format!("{}{}", self.base, nav.url))
552                })
553            });
554        // Drafts are dimmed, like they are on the site.
555        let draft = self.page.as_ref().is_some_and(|p| p.draft);
556        let title_style = if draft {
557            Style::new().fg(FAINT).bold()
558        } else {
559            Style::new().fg(TEXT).bold()
560        };
561        let mut block = Block::new()
562            .borders(Borders::ALL)
563            .padding(Padding::horizontal(1))
564            .border_style(Style::new().fg(BORDER));
565        if !named_in_nav {
566            block = block.title(
567                Line::from_iter(
568                    std::iter::once(Span::styled(
569                        if draft {
570                            format!(" Draft: {title} ")
571                        } else {
572                            format!(" {title} ")
573                        },
574                        title_style,
575                    ))
576                    .chain(
577                        self.page
578                            .as_ref()
579                            .and_then(|p| p.status.as_ref())
580                            .map(|s| Span::styled(format!("{s} "), Style::new().fg(MUTED))),
581                    ),
582                )
583                .centered(),
584            );
585        }
586        // Inside the corners, with a space either side.
587        let room = usize::from(column.width).saturating_sub(4);
588        if let Some(meta) = self.page.as_ref().and_then(|p| p.meta(room)) {
589            block = block.title_bottom(
590                Line::from(format!(" {meta} "))
591                    .style(Style::new().fg(MUTED))
592                    .centered(),
593            );
594        }
595        let inner = block.inner(column);
596        self.text = inner;
597        frame.render_widget(block, column);
598
599        let Some(page) = &mut self.page else {
600            frame.render_widget(Paragraph::new("Loading…").centered(), inner);
601            return;
602        };
603        frame.render_stateful_widget(MarkdownView::new(&self.theme), inner, &mut page.view);
604
605        // Links to drafts are faded, like drafts in the site's lists.
606        if let (Some(site), Some(rendered)) = (&self.site, page.view.rendered()) {
607            let scroll = page.view.scroll();
608            for link in &rendered.links {
609                let Target::Page { path, .. } = links::resolve(&link.url, &page.path, &self.origin)
610                else {
611                    continue;
612                };
613                let path = path.strip_prefix(&self.base).unwrap_or(&path);
614                if !site.is_draft(path) {
615                    continue;
616                }
617                for region in &link.regions {
618                    let Some(row) = region
619                        .line
620                        .checked_sub(scroll)
621                        .filter(|row| *row < usize::from(inner.height))
622                    else {
623                        continue;
624                    };
625                    let end = region.end.min(inner.width);
626                    let start = region.start.min(end);
627                    let rect = Rect::new(inner.x + start, inner.y + row as u16, end - start, 1);
628                    frame.buffer_mut().set_style(rect, Style::new().fg(FAINT));
629                }
630            }
631        }
632
633        let lines = page.view.rendered().map_or(0, |r| r.lines.len());
634        let max = lines.saturating_sub(usize::from(inner.height));
635        if max > 0 {
636            let mut state = ScrollbarState::new(max).position(page.view.scroll());
637            // Just a heavier stretch of the box's right border, with no
638            // track, so the border shows through.
639            frame.render_stateful_widget(
640                Scrollbar::new(ScrollbarOrientation::VerticalRight)
641                    .begin_symbol(None)
642                    .end_symbol(None)
643                    .track_symbol(None)
644                    .thumb_symbol("┃")
645                    .thumb_style(Style::new().fg(MUTED)),
646                column.inner(Margin::new(0, 1)),
647                &mut state,
648            );
649        }
650    }

653/// `markdown` with only the `<li data-tags="…">` items tagged `tag`, or
654/// all of them for `all`. `None` when no item has the tag, so the link is
655/// an ordinary one.
656fn filter_tagged(markdown: &str, tag: &str) -> Option<String> {
657    const ITEM: &str = "<li data-tags=\"";
658    let tagged = |tags: &str| tag == "all" || tags.split_whitespace().any(|t| t == tag);
659    let mut items = markdown.match_indices(ITEM).map(|(start, _)| {
660        let rest = &markdown[start + ITEM.len()..];
661        let tags = &rest[..rest.find('"').unwrap_or(0)];
662        let end = markdown[start..]
663            .find("</li>")
664            .map_or(markdown.len(), |e| start + e + "</li>".len());
665        (start, end, tagged(tags))
666    });
667    if !items.clone().any(|(_, _, keep)| keep) {
668        return None;
669    }
670    let mut out = String::with_capacity(markdown.len());
671    let mut at = 0;
672    for (start, end, keep) in &mut items {
673        if !keep {
674            out.push_str(&markdown[at..start]);
675            at = end;
676        }
677    }
678    out.push_str(&markdown[at..]);
679    Some(out)
680}

698/// Moves a project's status from before its title to after it, as small
699/// text. The site floats it to the right of the title, which a terminal
700/// can't, so `complete Epic UI` reads `Epic UI complete` instead.
701fn status_after_title(markdown: &str) -> String {
702    let mut out = String::with_capacity(markdown.len());
703    let mut status = None;
704    for line in markdown.lines() {
705        let trimmed = line.trim();
706        if trimmed.starts_with("<span class=\"status")
707            && trimmed.ends_with("</span>")
708            && let Some(start) = trimmed.find('>')
709        {
710            status = Some(trimmed[start + 1..trimmed.len() - 7].to_owned());
711            continue;
712        }
713        out.push_str(line);
714        if !trimmed.is_empty()
715            && let Some(status) = status.take()
716        {
717            out.push_str(&format!(" <small>{status}</small>"));
718        }
719        out.push('\n');
720    }
721    out
722}

724/// Starts the link in each draft's list item with `Draft: `, like a
725/// draft's own title. The site fades these items, which says the same thing
726/// in a browser.
727fn draft_prefixes(markdown: &str) -> String {
728    let mut out = String::with_capacity(markdown.len());
729    let mut rest = markdown;
730    while let Some(start) = rest.find("<li") {
731        let Some(end) = rest[start..].find('>').map(|i| start + i + 1) else {
732            break;
733        };
734        let tag = &rest[start..end];
735        out.push_str(&rest[..end]);
736        rest = &rest[end..];
737        let draft = tag
738            .split_once("class=\"")
739            .and_then(|(_, classes)| classes.split('"').next())
740            .is_some_and(|classes| classes.split_whitespace().any(|c| c == "draft"));
741        if draft
742            && let Some(a) = rest.find("<a ")
743            && let Some(close) = rest[a..].find('>').map(|i| a + i + 1)
744        {
745            out.push_str(&rest[..close]);
746            out.push_str("Draft: ");
747            rest = &rest[close..];
748        }
749    }
750    out.push_str(rest);
751    out
752}

758        _ => Target::Page {

760            anchor: None,

762    }

765/// Writes a `YYYY-MM-DD` date out like the site does, as `October 19, 2026`.
766fn long_date(date: &str) -> Option<String> {
767    const MONTHS: [&str; 12] = [
768        "January",
769        "February",
770        "March",
771        "April",
772        "May",
773        "June",
774        "July",
775        "August",
776        "September",
777        "October",
778        "November",
779        "December",
780    ];
781    let mut parts = date.splitn(3, '-');
782    let year = parts.next()?;
783    let month: usize = parts.next()?.parse().ok()?;
784    let day = parts.next()?.get(..2)?;
785    Some(format!(
786        "{} {day}, {year}",
787        MONTHS.get(month.checked_sub(1)?)?
788    ))
789}

791/// `markdown` without the title its layout puts at the top, or the line
792/// under it, which the page's box shows in its border instead. Everything
793/// else the layout adds, like a project's references, stays.
794fn without_title(markdown: &str) -> String {
795    // After the front matter, if there is any.
796    let start = markdown
797        .strip_prefix("---\n")
798        .and_then(|rest| rest.find("\n---\n"))
799        .map_or(0, |end| "---\n".len() + end + "\n---\n".len());
800    let (front, body) = markdown.split_at(start);
801    let Some(body) = [("<h1", "</h1>"), ("<h2", "</h2>")]
802        .into_iter()
803        .find_map(|(open, close)| skip_element(body.trim_start(), open, close))
804    else {
805        return markdown.to_owned();
806    };
807    let body = skip_element(body.trim_start(), "<p class=\"meta\"", "</p>").unwrap_or(body);
808    format!("{front}{body}")
809}

811/// What follows the element `html` starts with, if it starts with `open`.
812fn skip_element<'a>(html: &'a str, open: &str, close: &str) -> Option<&'a str> {
813    let end = html.strip_prefix(open)?.find(close)? + open.len();
814    Some(&html[end + close.len()..])
815}
