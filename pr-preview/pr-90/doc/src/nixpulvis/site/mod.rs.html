1//! The site's [`Theme`] for [cratesite], which builds `_site` from the
2//! pages compiled into this crate.
3//!
4//! cratesite reads the pages, renders their Markdown and writes the site.
5//! What's left is this site's own:
6//!
7//! - Pages with template tags, like [`ramblings`](crate::ramblings), are
8//!   askama templates, which can use the [`Site`] and its [`Data`].
9//! - Pages are wrapped in their [`Layout`], and then [`DefaultLayout`], with
10//!   the header and nav.
11//! - The fortunes get their `strfile` databases, and `site.json`, the
12//!   [`Index`], lists the pages for the terminal view.
13

30/// A page of this site.

33/// This site's own front matter, beside what cratesite reads.
34#[derive(Debug, Default, Deserialize)]
35#[serde(default)]
36pub struct Extra {
37    /// Moved off the Ramblings list, to its archive.
38    pub archived: bool,
39    /// Left out of every list.
40    pub hidden: bool,
41    /// A project's status, like `complete`.
42    pub status: Option<String>,
43    /// A project's links, listed under its title.
44    pub references: Vec<String>,
45    /// An image for link previews.
46    pub image: Option<String>,
47    /// A stylesheet just for this page.
48    pub style: Option<String>,
49    /// Scripts just for this page.
50    pub scripts: Vec<String>,
51}

53/// The site, its pages and its data.
54pub struct Site {
55    site: cratesite::Site<Extra>,
56    pub data: Data,
57    pub fortunes: Vec<Fortunes>,
58}

60impl Site {
61    pub const TITLE: &str = CONFIG.title;
62    pub const URL: &str = CONFIG.url;
63    pub const DESCRIPTION: &str = CONFIG.description;
64    pub const REPOSITORY: &str = "https://github.com/nixpulvis/nixpulvis.github.io";
65
66    /// Reads every page and data file, ready to [build](Theme::build).
67    pub fn load(options: Options) -> Result<Self, String> {
68        Ok(Self {
69            data: Data::load()?,
70            fortunes: Fortunes::read(&options.source.join("fortunes"))?,
71            site: cratesite::Site::load(&CONFIG, crate::SOURCES, options)?,
72        })
73    }
74
75    /// The index published as `site.json`, of the nav and every page in a
76    /// collection.
77    pub fn index(&self) -> Index {
78        let mut pages = Vec::new();
79        for collection in CONFIG.collections {
80            for page in self.collection(collection.label) {
81                pages.push(Entry {
82                    title: page.title(),
83                    date: page.date.map(|d| d.format("%Y-%m-%d").to_string()),
84                    collection: Some(collection.label.to_owned()),
85                    draft: page.front.draft,
86                    url: page.url.clone(),
87                });
88            }
89        }
90        Index {
91            title: Self::TITLE.to_owned(),
92            url: Self::URL.to_owned(),
93            baseurl: self.options.baseurl.clone(),
94            nav: self.data.nav.clone(),
95            pages,
96        }
97    }
98
99    /// The ramblings and mathematics, newest first, as they're listed on
100    /// the Ramblings page and in the feed.
101    pub fn ramblings(&self) -> Vec<&Page> {
102        let mut pages = self.collection("ramblings");
103        pages.extend(self.collection("mathematics"));
104        let today = self.time.date_naive();
105        pages.sort_by_key(|p| std::cmp::Reverse(p.date.unwrap_or(today)));
106        pages
107    }
108
109    /// One fortune from every year's, picked at random each build.
110    pub fn random_fortune(&self) -> &str {
111        let all: Vec<&String> = self.fortunes.iter().flat_map(|y| &y.entries).collect();
112        if all.is_empty() {
113            return "";
114        }
115        let i = self.time.timestamp_subsec_nanos() as usize % all.len();
116        all[i]
117    }
118
119    /// Every fortune, as a JSON array.
120    pub fn fortunes_json(&self) -> String {
121        let all: Vec<&String> = self.fortunes.iter().flat_map(|y| &y.entries).collect();
122        serde_json::to_string(&all).unwrap_or_default()
123    }
124}

134impl Theme for Site {
135    type Extra = Extra;
136
137    fn site(&self) -> &cratesite::Site<Extra> {
138        &self.site
139    }
140
141    fn site_mut(&mut self) -> &mut cratesite::Site<Extra> {
142        &mut self.site
143    }
144
145    fn template(&self, page: &Page) -> Option<Result<String, String>> {
146        templates::render(page.source.path, self, page).map(|r| r.map_err(|e| e.to_string()))
147    }
148
149    fn layout(&self, page: &Page, content: &str) -> Result<String, String> {
150        Layout::named(page.front.layout.as_deref()).wrap(self, page, content)
151    }
152
153    fn document(&self, page: &Page, content: &str) -> Result<String, String> {
154        use askama::Template as _;
155        DefaultLayout {
156            site: self,
157            page,
158            content,
159        }
160        .render()
161        .map_err(|e| e.to_string())
162    }
163
164    fn front_matter(&self, page: &Page) -> Yaml {
165        let title = page.title();
166        let date = page.date.map(|d| d.format("%Y-%m-%d").to_string());
167        // Only the layouts that show a word count.
168        let words = matches!(page.front.layout.as_deref(), Some("rambling" | "article"))
169            .then(|| page.words());
170        Yaml::new()
171            .string("title", (!title.is_empty()).then_some(title.as_str()))
172            .string("date", date.as_deref())
173            .string("description", page.front.description.as_deref())
174            .flag("draft", page.front.draft)
175            .string("status", page.front.extra.status.as_deref())
176            .list("references", &page.front.extra.references)
177            .number("words", words)
178            .string("collection", page.collection.map(|c| c.label))
179    }
180
181    fn files(&self) -> Result<Vec<(String, Vec<u8>)>, String> {
182        let mut files: Vec<(String, Vec<u8>)> = self
183            .fortunes
184            .iter()
185            .map(|year| (year.dat(), year.strfile()))
186            .collect();
187        let index = serde_json::to_string_pretty(&self.index()).map_err(|e| e.to_string())?;
188        files.push(("site.json".to_owned(), index.into_bytes()));
189        Ok(files)
190    }
191}

193/// What the templates ask of a page, beyond what cratesite gives it.
194pub trait Post {
195    /// `drafted` or `published`.
196    fn verb(&self) -> &'static str;
197
198    /// The status as a CSS class, like `partially-complete`.
199    fn status_class(&self) -> String;
200}
201
202impl Post for Page {
203    fn verb(&self) -> &'static str {
204        if self.front.draft {
205            "drafted"
206        } else {
207            "published"
208        }
209    }
210
211    fn status_class(&self) -> String {
212        self.front
213            .extra
214            .status
215            .as_deref()
216            .map(Name::slug)
217            .unwrap_or_default()
218    }
219}
