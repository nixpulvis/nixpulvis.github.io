19/// How long a notice shows before the key hints come back.
20const NOTICE_TIME: Duration = Duration::from_secs(3);
21
22/// Widest the page's text gets, for comfortable reading.
23const MAX_WIDTH: u16 = 84;
24
25impl App {
26    pub fn draw(&mut self, frame: &mut Frame) {
27        let area = frame.area();
28        frame.render_widget(
29            Block::new().style(Style::new().bg(BACKGROUND).fg(TEXT)),
30            area,
31        );
32        if self.starting() {
33            return;
34        }
35        let footer = self.footer(area.width);
36        let [header, body, bottom] = Layout::vertical([
37            Constraint::Length(1),
38            Constraint::Fill(1),
39            Constraint::Length(u16::from(footer.is_some())),
40        ])
41        .areas(area);
42        if let Some(footer) = footer {
43            frame.render_widget(
44                Paragraph::new(footer)
45                    .style(Style::new().fg(MUTED))
46                    .centered(),
47                bottom,
48            );
49        }
50        self.draw_header(frame, column(header));
51        self.draw_page(frame, column(body));
52    }
53
54    /// The line at the bottom. The page loading, messages and notices always
55    /// show. The key hints only show when they fit whole, and the page takes
56    /// their line when they don't.
57    fn footer(&mut self, width: u16) -> Option<String> {
58        if self
59            .notice
60            .as_ref()
61            .is_some_and(|(_, at)| at.elapsed() > NOTICE_TIME)
62        {
63            self.notice = None;
64        }
65        if let Some(loading) = &self.loading {
66            return Some(format!("Loading {}…", loading.path));
67        }
68        match (&self.message, &self.notice) {
69            (Some(message), _) | (None, Some((message, _))) => Some(message.clone()),
70            (None, None) => {
71                Some(self.hint()).filter(|h| Span::raw(h.as_str()).width() <= usize::from(width))
72            }
73        }
74    }
75
76    /// The key hints, with as many number keys as there are pages in the
77    /// nav.
78    fn hint(&self) -> String {
79        let pages = match self.nav().len().min(9) {
80            0 | 1 => String::new(),
81            n => format!(" · 1-{n} pages"),
82        };
83        let quit = match self.host {
84            Host::Terminal => "quit",
85            Host::Browser => "leave",
86        };
87        format!("j/k scroll · tab links · enter open · h back{pages} · q {quit}")
88    }

90    fn draw_header(&mut self, frame: &mut Frame, area: Rect) {
91        let current = self.page.as_ref().map_or("", |p| p.path.as_str());
92        let mut spans = vec![Span::raw(" ")];
93        let mut x = area.x + 1;
94        self.drawn.nav.clear();
95        for (i, item) in self.nav().into_iter().enumerate() {
96            let width = Span::raw(&item.label).width() as u16;
97            let style = if i == 0 {
98                Style::new().fg(TEXT).bold()
99            } else if same_page(current, &item.path) {
100                Style::new().fg(YELLOW).add_modifier(Modifier::UNDERLINED)
101            } else {
102                Style::new().fg(MUTED)
103            };
104            spans.push(Span::styled(item.label, style));
105            spans.push(Span::raw("  "));
106            self.drawn
107                .nav
108                .push((Rect::new(x, area.y, width, 1), item.path));
109            x += width + 2;
110        }
111        frame.render_widget(Line::from(spans), area);
112
113        // A way out for touch screens, with no keyboard for `q`.
114        self.drawn.close = Rect::new(area.right().saturating_sub(3), area.y, 3, 1);
115        frame.render_widget(
116            Line::from(" × ").style(Style::new().fg(MUTED)),
117            self.drawn.close,
118        );
119
120        #[cfg(feature = "native-links")]
121        {
122            let buffer = frame.buffer_mut();
123            for (rect, path) in &self.drawn.nav {
124                buffer.set_hyperlink(
125                    *rect,
126                    Some(&ratatui::buffer::Hyperlink::new(self.url(path))),
127                );
128            }
129            // A link, so it looks clickable, to this page as plain HTML. A
130            // plain click leaves, like `q`.
131            let link = ratatui::buffer::Hyperlink::new(self.url(current));
132            buffer.set_hyperlink(self.drawn.close, Some(&link));
133        }
134    }
135
136    fn draw_page(&mut self, frame: &mut Frame, column: Rect) {
137        let block = self.page_box(column.width);
138        let inner = block.inner(column);
139        self.drawn.text = inner;
140        frame.render_widget(block, column);
141
142        // Only when the first page failed to load, which the footer says.
143        let Some(page) = &mut self.page else {
144            return;
145        };
146        frame.render_stateful_widget(MarkdownView::new(&self.theme), inner, &mut page.view);
147        self.fade_draft_links(frame, inner);
148        self.draw_scrollbar(frame, column.inner(Margin::new(0, 1)), inner.height);
149    }

151    /// The box around the page, with its title on top and the line under a
152    /// post at the bottom, for a box `width` wide.
153    fn page_box(&self, width: u16) -> Block<'static> {
154        let block = Block::new()
155            .borders(Borders::ALL)
156            .padding(Padding::horizontal(1))
157            .border_style(Style::new().fg(BORDER));
158        let Some(page) = &self.page else {
159            return block;
160        };
161        // A page the nav names in words is already underlined there.
162        let named = self
163            .nav()
164            .iter()
165            .any(|item| item.names_page && same_page(&page.path, &item.path));
166        let block = if named {
167            block
168        } else {
169            block.title(page_title(page).centered())
170        };
171        // Inside the corners, with a space either side.
172        match page.meta(usize::from(width).saturating_sub(4)) {
173            Some(meta) => block.title_bottom(
174                Line::from(format!(" {meta} "))
175                    .style(Style::new().fg(MUTED))
176                    .centered(),
177            ),
178            None => block,
179        }
180    }
181
182    /// Fades the links to drafts, like drafts in the site's lists, on the
183    /// page drawn in `inner`.
184    fn fade_draft_links(&self, frame: &mut Frame, inner: Rect) {
185        let (Some(site), Some(page)) = (&self.site, &self.page) else {
186            return;
187        };
188        let Some(rendered) = page.view.rendered() else {
189            return;
190        };
191        let scroll = page.view.scroll();
192        for link in &rendered.links {
193            let Target::Page { path, .. } = Target::resolve(&link.url, &page.path, &self.origin)
194            else {
195                continue;
196            };
197            if !site.is_draft(path.strip_prefix(&self.base).unwrap_or(&path)) {
198                continue;
199            }
200            for region in &link.regions {
201                let Some(row) = region
202                    .line
203                    .checked_sub(scroll)
204                    .filter(|row| *row < usize::from(inner.height))
205                else {
206                    continue;
207                };
208                let end = region.end.min(inner.width);
209                let start = region.start.min(end);
210                let rect = Rect::new(inner.x + start, inner.y + row as u16, end - start, 1);
211                frame.buffer_mut().set_style(rect, Style::new().fg(FAINT));
212            }
213        }
214    }

216    /// Just a heavier stretch of the box's right border in `area`, with no
217    /// track, so the border shows through. Only there when the page is
218    /// longer than its `height`.
219    fn draw_scrollbar(&self, frame: &mut Frame, area: Rect, height: u16) {
220        let Some(page) = &self.page else {
221            return;
222        };
223        let lines = page.view.rendered().map_or(0, |r| r.lines.len());
224        let max = lines.saturating_sub(usize::from(height));
225        if max == 0 {
226            return;
227        }
228        let mut state = ScrollbarState::new(max).position(page.view.scroll());
229        frame.render_stateful_widget(
230            Scrollbar::new(ScrollbarOrientation::VerticalRight)
231                .begin_symbol(None)
232                .end_symbol(None)
233                .track_symbol(None)
234                .thumb_symbol("┃")
235                .thumb_style(Style::new().fg(MUTED)),
236            area,
237            &mut state,
238        );
239    }
240}

242/// A page's title for the top of its box, with a project's status after
243/// it. Drafts are dimmed, like they are on the site.
244fn page_title(page: &Page) -> Line<'static> {
245    let (title, style) = if page.draft {
246        (format!(" Draft: {} ", page.title), Style::new().fg(FAINT))
247    } else {
248        (format!(" {} ", page.title), Style::new().fg(TEXT))
249    };
250    let status = page
251        .status
252        .as_ref()
253        .map(|s| Span::styled(format!("{s} "), Style::new().fg(MUTED)));
254    Line::from_iter(std::iter::once(Span::styled(title, style.bold())).chain(status))
255}
256
257/// The middle of `area`, as wide as a page's box, so the nav lines up with it.
258fn column(area: Rect) -> Rect {
259    let width = area.width.min(MAX_WIDTH + 4);
260    Rect::new(
261        area.x + (area.width - width) / 2,
262        area.y,
263        width,
264        area.height,
265    )
266}
