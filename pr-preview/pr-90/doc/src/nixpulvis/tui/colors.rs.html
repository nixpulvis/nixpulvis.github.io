1//! The site's dark color scheme, from `css/style.css`, for the terminal.
2
3use ratatui::style::{Color, Modifier, Style};
4use ratmark::Theme;
5
6/// `--background`
7pub const BACKGROUND: Color = Color::Rgb(0x16, 0x16, 0x16);
8/// `--text`
9pub const TEXT: Color = Color::Rgb(0xca, 0xca, 0xca);
10/// `--muted`
11pub const MUTED: Color = Color::Rgb(0x99, 0x99, 0x99);
12/// `--border`
13pub const BORDER: Color = Color::Rgb(0x77, 0x77, 0x77);
14/// `--faint`
15pub const FAINT: Color = Color::Rgb(0x66, 0x66, 0x66);
16/// `--light`, behind code blocks.
17pub const LIGHT: Color = Color::Rgb(0x28, 0x28, 0x28);
18/// Between the background and `--light`, behind inline code. The HTML
19/// marks it with a font, which a terminal can't.
20pub const SHADE: Color = Color::Rgb(0x24, 0x24, 0x24);
21/// `--highlight`, behind highlighted code.
22pub const HIGHLIGHT: Color = Color::Rgb(0x35, 0x2d, 0x17);
23/// `--highlight-red`
24pub const HIGHLIGHT_RED: Color = Color::Rgb(0x3b, 0x1f, 0x1d);
25/// `--highlight-blue`
26pub const HIGHLIGHT_BLUE: Color = Color::Rgb(0x1c, 0x28, 0x40);
27/// `--highlight-edge`
28pub const YELLOW: Color = Color::Rgb(0xf0, 0xc9, 0x4a);
29/// `--highlight-blue-edge`
30pub const BLUE: Color = Color::Rgb(0x5a, 0x8e, 0xe0);
31/// `--highlight-purple-edge`
32pub const PURPLE: Color = Color::Rgb(0x9a, 0x6a, 0xd6);
33
34/// ratmark's theme in the site's colors. Like the HTML, it's mostly text
35/// color, with weight and underlines doing the work.
36pub fn theme() -> Theme {
37    let bold = Style::new().fg(TEXT).add_modifier(Modifier::BOLD);
38    Theme {
39        text: Style::new().fg(TEXT),
40        headings: [
41            bold.add_modifier(Modifier::UNDERLINED),
42            bold,
43            bold,
44            bold,
45            Style::new().fg(TEXT),
46            Style::new().fg(TEXT),
47        ],
48        link: Style::new().fg(BLUE).add_modifier(Modifier::UNDERLINED),
49        image: Style::new().fg(PURPLE),
50        // No color of its own, so code in a link stays the link's color.
51        code: Style::new().bg(SHADE),
52        code_block: Style::new().fg(TEXT).bg(LIGHT),
53        quote: Style::new().fg(MUTED),
54        quote_marker: Style::new().fg(FAINT),
55        list_marker: Style::new().fg(MUTED),
56        rule: Style::new().fg(FAINT),
57        math: Style::new().fg(TEXT),
58        footnote: Style::new().fg(MUTED),
59        table_border: Style::new().fg(FAINT),
60        table_header: bold,
61        small: Style::new().fg(MUTED),
62        // The backgrounds of highlighted code, as in the CSS, without the
63        // edges.
64        classes: vec![
65            (".highlight-block".into(), Style::new().bg(HIGHLIGHT)),
66            (
67                ".highlight-block.red".into(),
68                Style::new().bg(HIGHLIGHT_RED),
69            ),
70            (
71                ".highlight-block.other".into(),
72                Style::new().bg(HIGHLIGHT_BLUE),
73            ),
74            (".new-syntax".into(), Style::new().bg(HIGHLIGHT)),
75        ],
76        ..Theme::gruvbox()
77    }
78}
