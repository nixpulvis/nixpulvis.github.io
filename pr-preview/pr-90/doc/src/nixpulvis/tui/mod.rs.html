1//! Browse nixpulvis.com as a terminal app.
2//!
3//! The app reads the site's published Markdown and `site.json`, renders
4//! pages with ratmark, and draws them with Ratatui. It never does I/O itself.
5//! Methods return [`Effect`]s for the host to carry out, which keeps the
6//! same code working in a real terminal and in the browser.
7//!
8//! ```no_run
9//! # use nixpulvis::tui::{App, Host};
10//! let mut app = App::new(Host::Terminal, "https://nixpulvis.com", "/");
11//! let effects = app.start("/ramblings");
12//! // Carry out the effects, pass what they fetch back to the app, and
13//! // draw it and pass it input until it sends `Effect::Quit`.
14//! ```
15

25/// Where the app runs, which decides who keeps the history, and what `q`
26/// does.
27#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28pub enum Host {
29    /// A terminal, where the app is all there is. It keeps its own history,
30    /// and `q` quits.
31    Terminal,
32    /// A browser, with the app drawn over the site's own page. The browser
33    /// loads the site's other pages, like it would any link, and so it keeps
34    /// the history. `q` leaves the app for the page under it.
35    Browser,
36}
37
38/// Something for the host to do.
39#[derive(Debug, Clone, PartialEq, Eq)]
40pub enum Effect {
41    /// Fetch `site.json` from this path and pass it to [`App::site_loaded`].
42    FetchSite(String),
43    /// Fetch a page's Markdown from `source` and pass it to
44    /// [`App::page_loaded`] along with `path`.
45    FetchPage { path: String, source: String },
46    /// Open a link that isn't part of the site.
47    Open(String),
48    /// Load the site's page at this path, with any `#` section, and start
49    /// the app on it again. Only sent in a [`Host::Browser`].
50    Navigate(String),
51    /// Go back to the page before this one. Only sent in a
52    /// [`Host::Browser`].
53    Back,
54    /// Stop, or in a [`Host::Browser`], leave the app for the page under it.
55    Quit,
56}
57
58/// Input, already translated from whatever the host's events look like.
59#[derive(Debug, Clone, Copy, PartialEq, Eq)]
60pub enum Key {

63    PageUp,
64    PageDown,
65    Home,
